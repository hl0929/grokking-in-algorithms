BFS 迭代求深度
def max_depth(root):
if not root: return 0
queue = [root]
depth = 0
while queue:
depth += 1
next_level = []
for node in queue:
if node.left:
next_level.append(node.left)
if node.right:
next_level.append(node.right)
queue = next_level
return depth
