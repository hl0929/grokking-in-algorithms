def level_traversal(root):
result = []
queue = [root]
while queue:
part = []
level_queue = []
for node in queue:
part.append(node.val)
if node.left:
level_queue.append(node.left)
if node.right:
level_queue.append(node.right)
result.append(part)
queue = level_queue
return result
